% =========================================================
% Compatibility with existing chapter files
%
% Main chapters become article sections.
% Within each imported chapter:
%   \section       -> \subsection
%   \subsection    -> \subsubsection
%   \subsubsection -> \paragraph

% =========================================================

  \begingroup
    \let\section\ArticleSubsection
    \let\subsection\ArticleSubsubsection
    \let\subsubsection\ArticleParagraph
    \input{#1}%
    \par
  \endgroup
}
